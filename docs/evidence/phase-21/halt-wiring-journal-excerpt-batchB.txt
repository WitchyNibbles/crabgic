import { mkdtemp, rm } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { createJournalStore } from "@crabgic/journal";
import { buildJiraFieldDiffs, buildMaterialAmendmentSignal } from "@crabgic/gates";
import { createRunsRegistry, haltRunOnMaterialAmendment, transitionRun } from "@crabgic/supervisor";

const CHANGE_SET_ID = "44444444-4444-4444-8444-444444444444";
const REQUIREMENT_ID = "66666666-6666-4666-8666-666666666666";
const HALTED_RUN = "77777777-0000-4000-8000-00000000000a";
const CONTROL_RUN = "77777777-0000-4000-8000-00000000000b";

const journalDir = await mkdtemp(join(tmpdir(), "eo-halt-excerpt-batchB-"));
const store = createJournalStore({ journalDir });
const runs = createRunsRegistry();

async function driveToRunning(runId) {
  for (const to of ["awaiting_approval", "ready", "running"]) {
    await transitionRun({ journal: store, runs, runId, changeSetId: CHANGE_SET_ID, to });
  }
}
async function entries(runId, type) {
  const found = [];
  for await (const e of store.queryEntries({ type, runId })) found.push(e);
  return found;
}
const show = (label, value) => console.log(`\n--- ${label} ---\n${JSON.stringify(value, null, 2)}`);

// ============ LEG 1: a material remote amendment halts an in-flight run ============
await driveToRunning(HALTED_RUN);
console.log(`run ${HALTED_RUN} before the amendment: ${runs.get(HALTED_RUN).runState}`);

READING THE EXCERPT

  LEG 1 (the halted run, 77777777-...-00000000000a) is the artifact 21:153 names.
  Note four things in it:

    * seq 4 `run_transition {from: "running", to: "blocked"}` PRECEDES seq 5's
      `adjudication_decision`. That ordering is `stop-conditions.ts`'s own LOW L7
      repair — the decision record is written only once the halt has genuinely
      taken effect — and it survives the new wrapper.
    * the decision's `rationale` names the KIND verbatim:
      `stop condition "material_amendment"`. The criterion names the kind, so the
      rationale content is asserted rather than the mere existence of a decision;
      probe M4 (see halt-wiring-probes-batchB.txt) measures that this is
      load-bearing.
    * "before `final_verifying`" is proven against the state machine, not a flag:
      resuming the halted run to `verifying` is refused with
      `IllegalTransitionError`, and `blocked` is absorbing, so the run provably
      can never reach `final_verifying` afterwards.
    * `materialFields: ["description"]` comes from 21's real allow-list, not from
      the driver.

  LEG 2 is the mandatory does-not-halt control, on a SECOND run
  (77777777-...-00000000000b). Only a non-tracked custom field moved
  (`customfield_10200`), so 21's conservative classifier returns
  `{material: false, materialFields: []}`, the wiring returns `{halted: false}`,
  the run stays `running`, ZERO `adjudication_decision` entries exist for it, and
  it walks on cleanly to `final_verifying`. Without this leg, a wiring
  implemented as an unconditional `haltOnStopCondition(...)` would satisfy leg 1
  identically.
