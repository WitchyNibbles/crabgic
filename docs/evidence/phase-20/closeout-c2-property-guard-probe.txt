# | import fc from "fast-check";
# | import { hashCanonicalFields } from "@crabgic/connectors-grafana";
# | import { resolveOptimisticConcurrencyConflict } from "@crabgic/connectors-grafana";
# | 
# | class SimulatedRemoteResource {
# |   #revision = 0;
# |   #content;
# |   constructor(initialContent) {
# |     this.#content = initialContent;
# |   }
# |   read() {
# |     return { revision: String(this.#revision), contentHash: hashCanonicalFields({ content: this.#content }) };
# |   }
# |   tryWrite(expectedRevision, newContent) {
# |     if (expectedRevision !== String(this.#revision)) return { ok: false, status: 409 };
# |     this.#content = newContent;
# |     this.#revision += 1;
# |     return { ok: true };
# |   }
# | }

# | function simulateWriterAttempt(remote, baselineRevision, baselineContentHash, desiredContent) {
# |   const first = remote.tryWrite(baselineRevision, desiredContent);
# |   if (first.ok) return "applied";

# |   const resolution = resolveOptimisticConcurrencyConflict({ baselineContentHash, currentRemote });
# |   if (resolution.kind === "block") return "blocked";
# |   const retried = remote.tryWrite(resolution.freshRevision, desiredContent);
# |   return retried.ok ? "rebased-and-applied" : "blocked";
