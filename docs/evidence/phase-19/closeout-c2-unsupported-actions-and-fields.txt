  JiraPlanPayloadRegistry,
} from "@crabgic/connectors-jira";
import { GatewayHttpClient, createFakeProviderTransport } from "@crabgic/gateway";
import { buildExternalConnection } from "@crabgic/testkit";
import { toADF } from "@crabgic/renderer";

const BASE_URL = "https://dc-field-probe.invalid";
const ENVELOPE_ID = "ffffffff-ffff-4fff-8fff-ffffffffffff";
const FULL_ACTIONS = [
  "issue.create","issue.update","issue.transition","issue.link","issue.rank",
  "issue.bulkUpdate","issue.bulkTransition","comment.create","comment.update",
  "worklog.create","attachment.upload","board.create","board.update",
  "sprint.create","sprint.start","sprint.complete","sprint.moveIssues",
];

const connection = buildExternalConnection({
  provider: "jira-datacenter", deploymentType: "datacenter", baseUrl: BASE_URL,
});
const fake = createFakeProviderTransport({ responses: [] });
const httpClient = new GatewayHttpClient({
  allowlist: { allowedSchemes: ["https:"], allowedOrigins: [new URL(BASE_URL).origin] },
  resolveHostAddresses: async () => ["203.0.113.230"],
  sendRequest: fake.send,
  sleep: async () => undefined,
});
const client = createJiraDatacenterResourceClient({
  ctx: { connection, httpClient, authHeaderProvider: async () => ({ authorization: "Bearer x" }) },
  fieldMetadataIndex: buildFieldMetadataIndex([]),
  payloadRegistry: new JiraPlanPayloadRegistry(),
  dcFeatures: { edition: "10.3", availableActions: FULL_ACTIONS, availableFields: "discovered-only" },
});

function report(label, fn) {
  try {
    fn();
    console.log(`${label}: NO THROW — the write was accepted`);
  } catch (err) {
    console.log(
      `${label}: kind=${JSON.stringify(err.kind)} provider=${JSON.stringify(err.provider)} retryable=${JSON.stringify(err.retryable)}`,
    );
    console.log(`${label}: message=${JSON.stringify(err.message)}`);
  }
}

console.log("dcFeatures.availableFields is the literal:", JSON.stringify("discovered-only"));

  client.issues.planUpdate("PROJ-1", "rev-1", { customfield_99999: "x" }, ENVELOPE_ID),
);
report("issues.planCreate  customfield_99999 (undiscovered)", () =>
  client.issues.planCreate(
    { projectKeyOrId: "PROJ", issueType: "Story", summaryAdf: toADF("s"), fields: { customfield_99999: "x" } },
    ENVELOPE_ID,
  ),
);
report("issues.planBulkUpdate customfield_99999 (undiscovered)", () =>
  client.issues.planBulkUpdate(["PROJ-1"], { customfield_99999: "x" }, ENVELOPE_ID),
);
